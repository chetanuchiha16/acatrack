-- table: mentor_students
CREATE TABLE mentor_students (
	id INTEGER NOT NULL, 
	mentor_id INTEGER NOT NULL, 
	student_usn VARCHAR(50) NOT NULL, 
	PRIMARY KEY (id), 
	FOREIGN KEY(mentor_id) REFERENCES mentors (id), 
	FOREIGN KEY(student_usn) REFERENCES student_emails (usn)
);

-- table: mentors
CREATE TABLE mentors (
	id INTEGER NOT NULL, 
	name VARCHAR(100) NOT NULL, 
	PRIMARY KEY (id), 
	UNIQUE (name)
);

-- table: student_emails
CREATE TABLE student_emails (
	id INTEGER NOT NULL, 
	usn VARCHAR(50) NOT NULL, 
	name VARCHAR(100) NOT NULL, 
	parent_email VARCHAR(100) NOT NULL, 
	student_email VARCHAR(100) NOT NULL, 
	parent_phno VARCHAR(100), 
	student_phno VARCHAR(100), 
	PRIMARY KEY (id), 
	UNIQUE (usn)
);

-- table: teachers
CREATE TABLE teachers (
	username VARCHAR(10) NOT NULL, 
	name VARCHAR(100), 
	password VARCHAR(128), 
	PRIMARY KEY (username), 
	UNIQUE (username)
);

-- table: users
CREATE TABLE users (
	username VARCHAR(10) NOT NULL, 
	name VARCHAR(100), 
	password VARCHAR(128), 
	PRIMARY KEY (username), 
	UNIQUE (username)
);
